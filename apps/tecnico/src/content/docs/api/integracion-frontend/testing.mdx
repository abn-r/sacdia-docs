## Next.js Tests (Vitest + MSW + RTL)

El admin usa **Vitest** como runner y **MSW** (Mock Service Worker) para interceptar fetch. No usamos Jest ni Playwright.

```typescript
// tests/setup.ts
import { afterAll, afterEach, beforeAll } from 'vitest';
import { setupServer } from 'msw/node';
import { http, HttpResponse } from 'msw';

export const server = setupServer(
  http.get('*/api/v1/clubs', () =>
    HttpResponse.json({ status: 'success', data: [{ club_id: 1, name: 'Test Club' }] }),
  ),
);

beforeAll(() => server.listen({ onUnhandledRequest: 'error' }));
afterEach(() => server.resetHandlers());
afterAll(() => server.close());
```

```typescript
// tests/hooks/use-clubs.test.tsx
import { describe, it, expect } from 'vitest';
import { renderHook, waitFor } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { useClubs } from '@/hooks/use-clubs';

function wrapper({ children }: { children: React.ReactNode }) {
  const qc = new QueryClient({ defaultOptions: { queries: { retry: false } } });
  return <QueryClientProvider client={qc}>{children}</QueryClientProvider>;
}

describe('useClubs', () => {
  it('returns clubs from API', async () => {
    const { result } = renderHook(() => useClubs(), { wrapper });
    await waitFor(() => expect(result.current.isSuccess).toBe(true));
    expect(result.current.data).toEqual([{ club_id: 1, name: 'Test Club' }]);
  });
});
```

## Flutter Tests (Mockito + Dio mock)

```dart
// test/features/clubs/data/datasources/clubs_remote_datasource_test.dart
@GenerateMocks([Dio])
void main() {
  late ClubsRemoteDataSource dataSource;
  late MockDio mockDio;

  setUp(() {
    mockDio = MockDio();
    dataSource = ClubsRemoteDataSource(mockDio);
  });

  test('returns list of clubs on success', () async {
    when(mockDio.get('/clubs')).thenAnswer(
      (_) async => Response(
        data: { 'status': 'success', 'data': [{ 'club_id': 1, 'name': 'Test Club' }] },
        statusCode: 200,
        requestOptions: RequestOptions(path: '/clubs'),
      ),
    );

    final result = await dataSource.getClubs();
    expect(result.length, 1);
    expect(result.first.name, 'Test Club');
  });

  test('throws ServerException on 500', () async {
    when(mockDio.get('/clubs')).thenThrow(
      DioException(
        requestOptions: RequestOptions(path: '/clubs'),
        response: Response(statusCode: 500, requestOptions: RequestOptions(path: '/clubs')),
      ),
    );

    expect(() => dataSource.getClubs(), throwsA(isA<ServerException>()));
  });
}
```
